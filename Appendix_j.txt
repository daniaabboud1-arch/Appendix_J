import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize


# ============================================================
# Parameters
# ============================================================

P_MIN, P_MAX = 0.01, 0.99
N_GRID = 50
N_STARTS = 10
SEED = 12345
EPS = 1e-12

# Numerical tolerance for the Markov-point tests
MARKOV_TOL = 1e-10


# ============================================================
# Admissibility
# ============================================================

def admissible(p, t):
    """
    Check whether (p,t) belongs to the admissible parameter domain:
    
        -min(p^2,(1-p)^2) <= t <= p(1-p).
    """

    return (
        -min(p**2, (1-p)**2) - EPS
        <= t
        <= p * (1-p) + EPS
    )


# ============================================================
# Joint distribution
# ============================================================

def joint_distribution(p, t):
    """
    Construct P_{BCR} for the binary near-Markov model.
    """

    s = 1 - p

    A = np.array([
        [s**2 + t, p*s - t],
        [p*s - t, p**2 + t]
    ])

    P = np.zeros((2, 2, 2))

    for b in range(2):
        for x in range(2):
            for y in range(2):
                P[b, b ^ y, b ^ x] += 0.5 * A[x, y]

    return P


# ============================================================
# Conditional mutual information
# ============================================================

def cmi(p, t):
    """
    Exact conditional mutual information:
    
        I(C:R|B) = I(X:Y).
    """

    s = 1 - p

    terms = [
        (s**2 + t, s**2),
        (p*s - t, p*s),
        (p*s - t, p*s),
        (p**2 + t, p**2)
    ]

    return sum(
        a * np.log2(a / b)
        for a, b in terms
        if a > EPS and b > EPS
    )


# ============================================================
# Natural recovery fidelity
# ============================================================

def F_nat(p, t):
    """
    Fidelity of the natural classical recovery.
    """

    s = 1 - p

    a = max(s**2 + t, 0.0)
    b = max(p*s - t, 0.0)
    c = max(p**2 + t, 0.0)

    return (
        np.sqrt(a * s**2)
        + 2 * np.sqrt(b * p*s)
        + np.sqrt(c * p**2)
    )


# ============================================================
# B-preserving optimum
# ============================================================

def A_matrix(p, t):
    """
    Matrix entering the analytic B-preserving optimum.
    """

    P = joint_distribution(p, t)
    R = P.sum(axis=1)

    A = np.zeros((2, 2))

    for b in range(2):
        for c in range(2):
            for r in range(2):

                v = P[b, c, r] * R[b, r]

                if v > 0:
                    A[b, c] += np.sqrt(v)

    return A


def F_opt_B_pres(p, t):
    """
    Exact optimal fidelity within the B-preserving
    classical recovery class.
    """

    A = A_matrix(p, t)

    return sum(
        np.sqrt(np.sum(A[b]**2))
        for b in range(2)
    )


# ============================================================
# Fidelity for a general recovery kernel
# ============================================================

def fidelity(P, K):
    """
    Fidelity between the original distribution P and
    the recovered distribution generated by K.
    """

    R = P.sum(axis=1)

    Q = np.zeros((2, 2, 2))

    for b in range(2):
        for tb in range(2):
            for c in range(2):
                for r in range(2):

                    Q[tb, c, r] += R[b, r] * K[b, tb, c]

    return np.sqrt(np.maximum(P * Q, 0.0)).sum()


# ============================================================
# Unrestricted numerical optimization
# ============================================================

def F_opt_all(
    p,
    t,
    n_starts=N_STARTS,
    seed=SEED
):
    """
    Numerically optimize the unrestricted classical recovery
    kernel using SLSQP.

    Returns
    -------
    best_fidelity : float
        Best successful fidelity, or np.nan if every optimization
        attempt fails.

    success : bool
        True if at least one optimization run succeeds.
    """

    P = joint_distribution(p, t)

    rng = np.random.default_rng(seed)

    best = -np.inf
    success_found = False

    constraints = [
        {
            "type": "eq",
            "fun": lambda x: np.sum(x[:4]) - 1
        },
        {
            "type": "eq",
            "fun": lambda x: np.sum(x[4:]) - 1
        }
    ]

    for _ in range(n_starts):

        x0 = np.concatenate([
            rng.dirichlet(np.ones(4)),
            rng.dirichlet(np.ones(4))
        ])

        result = minimize(
            lambda x: -fidelity(
                P,
                x.reshape(2, 2, 2)
            ),
            x0,
            method="SLSQP",
            bounds=[(0, 1)] * 8,
            constraints=constraints,
            options={
                "maxiter": 5000,
                "ftol": 1e-10,
                "disp": False
            }
        )

        # ----------------------------------------------------
        # IMPORTANT:
        # An optimization failure is NOT replaced by another
        # fidelity value. It remains a failure.
        # ----------------------------------------------------

        if result.success and np.isfinite(result.fun):

            current_fidelity = -result.fun

            if np.isfinite(current_fidelity):

                success_found = True
                best = max(best, current_fidelity)

    if success_found and np.isfinite(best):
        return best, True

    return np.nan, False


# ============================================================
# Fawzi-Renner bound
# ============================================================

def F_FR(p, t):
    """
    Fawzi-Renner lower bound:
    
        F_FR = 2^{-I(C:R|B)/2}.
    """

    return 2 ** (-cmi(p, t) / 2)


# ============================================================
# Mandatory test at the Markov point t = 0
# ============================================================

print("=" * 70)
print("MARKOV-POINT TEST: t = 0")
print("=" * 70)

markov_failures = 0

for p in np.linspace(P_MIN, P_MAX, N_GRID):

    t = 0.0

    I = cmi(p, t)
    F_n = F_nat(p, t)
    F_fr = F_FR(p, t)
    F_b = F_opt_B_pres(p, t)

    # The unrestricted optimization is numerical.
    F_a, success = F_opt_all(
        p,
        t,
        n_starts=100,
        seed=SEED
    )

    # Check the exact/expected Markov identities
    tests_pass = (
        abs(I) < MARKOV_TOL
        and abs(F_n - 1.0) < MARKOV_TOL
        and abs(F_fr - 1.0) < MARKOV_TOL
        and abs(F_b - 1.0) < MARKOV_TOL
        and success
        and abs(F_a - 1.0) < MARKOV_TOL
    )

    if not tests_pass:
        markov_failures += 1

        print(
            f"Markov-point test failed at p = {p:.6f}"
        )
        print(f"    I(C:R|B)       = {I:.15f}")
        print(f"    F_nat           = {F_n:.15f}")
        print(f"    F_FR            = {F_fr:.15f}")
        print(f"    F_opt^(B-pres)  = {F_b:.15f}")
        print(f"    F_opt^(all)     = {F_a:.15f}")
        print(f"    optimization success = {success}")


if markov_failures == 0:
    print("Markov-point tests: PASSED")
    print(
        "For all tested p values, t = 0 gives "
        "I(C:R|B) = 0 and all fidelities equal 1 "
        "within the numerical tolerance."
    )
else:
    print(
        f"Markov-point tests: {markov_failures} FAILURE(S)"
    )

print("=" * 70)
print()

# ============================================================
# Parameter grid
# ============================================================

p_values = np.linspace(
    P_MIN,
    P_MAX,
    N_GRID
)

t_values = np.linspace(
    -0.25,
    0.25,
    N_GRID
)

shape = (
    N_GRID,
    N_GRID
)


# ============================================================
# Arrays
# ============================================================

I_grid = np.full(
    shape,
    np.nan
)

F_nat_grid = np.full(
    shape,
    np.nan
)

F_FR_grid = np.full(
    shape,
    np.nan
)

F_B_grid = np.full(
    shape,
    np.nan
)

F_all_grid = np.full(
    shape,
    np.nan
)

# True only when an admissible unrestricted optimization fails
optimization_failure_mask = np.zeros(
    shape,
    dtype=bool
)


# ============================================================
# Counters
# ============================================================

admissible_count = 0
inadmissible_count = 0

# B-preserving fidelity is analytic, so there is no SLSQP
# optimization failure for this quantity.
failure_bpres = 0

# Unrestricted numerical optimization failures
failure_all = 0


# ============================================================
# Main numerical computation
# ============================================================

print("=" * 70)
print("COMPUTING QUANTITIES")
print("=" * 70)
print()

for i, p in enumerate(p_values):

    print(
        f"Row {i + 1}/{N_GRID}",
        end="\r"
    )

    for j, t in enumerate(t_values):

        # ----------------------------------------------------
        # Admissibility check
        # ----------------------------------------------------

        if not admissible(p, t):

            inadmissible_count += 1

            continue

        admissible_count += 1

        # ----------------------------------------------------
        # Exact quantities
        # ----------------------------------------------------

        I_grid[i, j] = cmi(p, t)

        F_nat_grid[i, j] = F_nat(p, t)

        F_FR_grid[i, j] = F_FR(p, t)

        F_B_grid[i, j] = F_opt_B_pres(p, t)

        # ----------------------------------------------------
        # Unrestricted numerical optimization
        # ----------------------------------------------------

        F_all, success = F_opt_all(
            p,
            t,
            N_STARTS,
            SEED + i * N_GRID + j
        )

        if success and np.isfinite(F_all):

            F_all_grid[i, j] = F_all

        else:

            # IMPORTANT:
            # Keep the optimization failure as NaN.
            # Do NOT replace it with F_B or F_nat.
            F_all_grid[i, j] = np.nan

            optimization_failure_mask[i, j] = True

            failure_all += 1


print()
print()
print("=" * 70)
print("COMPUTATIONS COMPLETED")
print("=" * 70)
print()

# ============================================================
# Heatmap function
# ============================================================

def save_heatmap(
    Z,
    title,
    name,
    vmin=None,
    vmax=None,
    show_failures=True
):
    """
    Save heatmap as PNG and PDF.

    Inadmissible points remain NaN and therefore appear blank.

    Unrestricted optimization failures are marked separately
    with black 'x' symbols.
    """

    P, T = np.meshgrid(
        p_values,
        t_values,
        indexing="ij"
    )

    fig, ax = plt.subplots(
        figsize=(8, 6)
    )

    image = ax.pcolormesh(
        P,
        T,
        Z,
        shading="auto",
        vmin=vmin,
        vmax=vmax
    )

    ax.set_xlabel(
        r"$p$"
    )

    ax.set_ylabel(
        r"$t$"
    )

    ax.set_title(
        title
    )

    fig.colorbar(
        image,
        ax=ax
    )

    # --------------------------------------------------------
    # Mark optimization failures separately.
    #
    # Inadmissible points are blank (NaN).
    # Optimization failures are marked with x.
    # --------------------------------------------------------

    if show_failures and np.any(
        optimization_failure_mask
    ):

        failure_p = P[
            optimization_failure_mask
        ]

        failure_t = T[
            optimization_failure_mask
        ]

        ax.scatter(
            failure_p,
            failure_t,
            marker="x",
            s=25,
            linewidths=1.2,
            label="optimization failure"
        )

        ax.legend(
            loc="best"
        )

    fig.tight_layout()

    # Save both formats
    for extension in ["png", "pdf"]:

        fig.savefig(
            f"{name}.{extension}",
            dpi=300,
            bbox_inches="tight"
        )

    plt.close(fig)
